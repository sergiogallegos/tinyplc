cmake_minimum_required(VERSION 3.28)
project(toyplc VERSION 0.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(TOYPLC_BLINK "Build the standalone M0 STM32 blink" OFF)
if(TOYPLC_BLINK)
  add_executable(blink port/nucleo_f446re/blink.c port/nucleo_f446re/startup.c)
  target_compile_options(blink PRIVATE -Wall -Wextra -Werror -ffreestanding -fno-builtin -ffunction-sections -fdata-sections)
  target_link_options(blink PRIVATE -nostdlib -Wl,--gc-sections -T${CMAKE_CURRENT_SOURCE_DIR}/port/nucleo_f446re/stm32f446re.ld -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/blink.map)
  set_target_properties(blink PROPERTIES SUFFIX .elf)
else()
  add_executable(toyplc-sim sim/main.c)
  target_include_directories(toyplc-sim PRIVATE core/include)
  target_compile_definitions(toyplc-sim PRIVATE _POSIX_C_SOURCE=200809L)
  target_compile_options(toyplc-sim PRIVATE -Wall -Wextra -Wpedantic -Werror)
  enable_testing()
  add_test(NAME sim_smoke COMMAND toyplc-sim)
  set_tests_properties(sim_smoke PROPERTIES PASS_REGULAR_EXPRESSION "M0 scans=5 period_ms=10 output=1")
endif()
