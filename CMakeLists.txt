cmake_minimum_required(VERSION 3.28)
project(tinyplc VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(TINYPLC_BLINK "Build the standalone M0 STM32 blink" OFF)
if(TINYPLC_BLINK)
  add_executable(blink port/nucleo_f446re/blink.c port/nucleo_f446re/startup.c)
  target_compile_options(blink PRIVATE -Wall -Wextra -Werror -ffreestanding -fno-builtin -ffunction-sections -fdata-sections)
  target_link_options(blink PRIVATE -nostdlib -Wl,--gc-sections -T${CMAKE_CURRENT_SOURCE_DIR}/port/nucleo_f446re/stm32f446re.ld -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/blink.map)
  set_target_properties(blink PROPERTIES SUFFIX .elf)
else()
  option(TINYPLC_SANITIZE "Enable host address and undefined-behavior sanitizers" OFF)
  option(TINYPLC_THREAD_SANITIZE "Enable host thread sanitizer" OFF)
  if(TINYPLC_SANITIZE AND TINYPLC_THREAD_SANITIZE)
    message(FATAL_ERROR "Use separate build directories for address and thread sanitizers")
  endif()
  if(TINYPLC_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
  if(TINYPLC_THREAD_SANITIZE)
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
  endif()
  add_library(tinyplc-core core/src/image.c core/src/tags.c core/src/vm.c core/src/runtime.c)
  target_include_directories(tinyplc-core PUBLIC core/include)
  target_compile_options(tinyplc-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_executable(tinyplc-sim sim/main.c)
  target_link_libraries(tinyplc-sim PRIVATE tinyplc-core)
  target_compile_definitions(tinyplc-sim PRIVATE _POSIX_C_SOURCE=200809L)
  target_compile_options(tinyplc-sim PRIVATE -Wall -Wextra -Wpedantic -Werror)
  enable_testing()
  add_test(NAME sim_smoke COMMAND tinyplc-sim)
  set_tests_properties(sim_smoke PROPERTIES PASS_REGULAR_EXPRESSION "M1 scans=5 period_ms=10 LED=1 N=2 generation=1")
  find_package(Threads REQUIRED)
  add_executable(core-tests tests/native/test_core.c)
  target_link_libraries(core-tests PRIVATE tinyplc-core Threads::Threads)
  target_compile_options(core-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME core_tests COMMAND core-tests)
  set_tests_properties(core_tests PROPERTIES TIMEOUT 30)
endif()
